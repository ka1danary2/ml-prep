```python
class Solution:
    def minSubArrayLen(self, target: int, nums: List[int]) -> int:
        answer_len = float('inf')
        left = 0
        temp = 0

        for right in range(len(nums)):
            item = nums[right]
            temp += item

            while temp >= target:
                answer_len = min(answer_len, right - left + 1)
                temp -= nums[left]
                left += 1
            
        if answer_len == float('inf'):
            return 0
        return answer_len
```
