Вот мое почти верное решение, потому что слишком долго работает (из-за spread оператора)

```python
class MinStack:
    def __init__(self):
        self.stack = []
        self.min_val = []
        
    def push(self, val: int) -> None:
        self.stack.append(val)

        if len(self.min_val) == 0:
            self.min_val.append(val)
            return None

        if val < self.min_val[-1]:
           self.min_val.append(val)
           return None
        else:
            self.min_val.append(self.min_val[-1])
            return None

        return None
        
    def pop(self) -> None:
        self.stack.pop()
        self.min_val.pop()

        return None
        
    def top(self) -> int:
        copy = [*self.stack]

        return copy[-1]

    def getMin(self) -> int:
        copy = [*self.min_val]

        return copy[-1]

# Your MinStack object will be instantiated and called as such:
# obj = MinStack()
# obj.push(val)
# obj.pop()
# param_3 = obj.top()
# param_4 = obj.getMin()
```

```python
class MinStack:
    def __init__(self):
        self.stack = []
        self.minStack = []

    def push(self, val: int) -> None:
        self.stack.append(val)

        if len(self.minStack) == 0:
            self.minStack.append(val)
        else:
            currentMin = min(val, self.minStack[-1])
            self.minStack.append(currentMin)

    def pop(self) -> None:
        self.stack.pop()
        self.minStack.pop()

    def top(self) -> int:
         return self.stack[-1]

    def getMin(self) -> int:
        return self.minStack[-1]

# Your MinStack object will be instantiated and called as such:
# obj = MinStack()
# obj.push(val)
# obj.pop()
# param_3 = obj.top()
# param_4 = obj.getMin()
```
