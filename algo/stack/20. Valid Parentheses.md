class Solution:
    def isValid(self, s: str) -> bool:
        stack = []

        open_operations = ['(', '{', '[']
        close_operations = [')', '}', ']']

        for index, char in enumerate(s):
            if char in open_operations:
                stack.append(char)
                continue

            if char in close_operations:

                if(len(stack) == 0):
                    return False

                left_operand = stack.pop()

                if left_operand == '(' and char == ')':
                    continue
                if left_operand == '[' and char == ']':
                    continue
                if left_operand == '{' and char == '}':
                    continue
            return False

        if len(stack) != 0:
            return False

        return True

class Solution:
    def isValid(self, s: str) -> bool:
        openb = ['(', '{', '[']
        closeb = [ ')', '}', ']']

        stack = []

        for char in s:
            if char in openb:
                stack.append(char)
                continue
            if char in closeb and len(stack) > 0:
                if char == ')' and stack[-1] == '(':
                    stack.pop()
                    continue
                if char == '}' and stack[-1] == '{':
                    stack.pop()
                    continue
                if char == ']' and stack[-1] == '[':
                    stack.pop()
                    continue
                else:
                    return False
            else:
                return False
        if len(stack) == 0:
            return True
        return False
