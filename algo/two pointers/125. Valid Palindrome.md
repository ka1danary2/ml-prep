class Solution:
    def isPalindrome(self, s: str) -> bool:

        s_clear = ''.join(c for c in s if c.isalnum()).lower()

        pointer1 = 0
        pointer2 = len(s_clear) - 1

        while(pointer1 < pointer2):
            if s_clear[pointer1] != s_clear[pointer2]:
                return False
            pointer1 += 1
            pointer2 -= 1
        return True
