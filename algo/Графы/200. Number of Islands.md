Ссылка на задачу - https://leetcode.com/problems/number-of-islands/

Решается через дфс [[1 dfs (обойти все варианты)]] . То есть мы запускаем дфс из всех вершин, исключая те, которые мы обошли и обходим весь остров.

Мое решение
```python
class Solution:

    def dfs(self, grid, visited, start, n, m):
        visited[start] = 1

        i, j = start
        operation = [(-1,0), (1,0), (0,-1), (0,1)]
        
        for oper in operation:
            a, b = oper
            
            if 0 <= i + a < n and 0 <= j + b < m and grid[i + a][j + b] == "1" and visited[(i + a, j + b)] == 0:
                self.dfs(grid, visited, (i + a, j + b), n, m)


        return visited
    
    def numIslands(self, grid: List[List[str]]) -> int:

        answerCount = 0

        rows = len(grid)
        cols = len(grid[0])

        visited = {(i, j):  0 for i in range(rows) for j in range(cols)}
        

        for i in range(rows):
            for j in range(cols):
                if grid[i][j] == "1" and visited[(i, j)] == 0:
                    self.dfs(grid, visited, (i, j), rows, cols)
                    answerCount += 1

        return answerCount
```
