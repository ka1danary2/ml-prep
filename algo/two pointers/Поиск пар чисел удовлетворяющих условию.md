def twoPointers(nums, k):
	left = 0
	answer = 0
	for right in range(len(nums)):
		item = nums[right]
		
		while right < len(nums) and nums[right] - nums[left] <= k:
			answer += 1
			left += 1
		answer = len(nums) - right
	
	return answer
