```python
def bestTeamSum(players):
	bestsum = 0
	nowsum = 0
	
	last = 0
	
	for first in range(len(players)): 
		# длина, может команда быть из 1 человека и это не подойдет, и сумма 
		while last < len(players) and (last == first or players[first] + players[first + 1] >= players[last]):
			nowsum += players[last]
			last += 1
		bestsum = max(bestsum, nowsum)
		nowsum -= players[first]
	return bestsum
```

```python
def findMaximumSizeSubArray(nums: List[int]):
	sum = 0
	maxSum = 0
	
	right = len(nums) - 1
	left = len(nums) - 1

	while left >= 0 :
		if left == right:
			sum += nums[left]
			left -= 1
			
		elif right - left + 1 == 2:
			sum += nums[left]
			left -= 1
			
		elif nums[right] <= nums[left] + nums[left + 1]:
			sum += nums[left]
			left -= 1

		else:
			sum -= nums[right]
			right -= 1
		
		maxSum = max(maxSum, sum)
	return maxSum
```
