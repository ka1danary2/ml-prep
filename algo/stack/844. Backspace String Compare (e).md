class Solution:
    def backspaceCompare(self, s: str, t: str) -> bool:
        stack_s = []
        stack_t = []

        delete = '#'

        for char in s:
            if char == delete and len(stack_s) > 0:
                stack_s.pop()
                continue

            if char == delete and len(stack_s) == 0:
                continue

            stack_s.append(char)

        for char in t:
            if char == delete and len(stack_t) > 0:
                stack_t.pop()
                continue

            if char == delete and len(stack_t) == 0:
                continue

            stack_t.append(char)
        
        if stack_t == stack_s:
            return True
        return False
