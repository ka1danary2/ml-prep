Решение

Это дейкста, но в словаре тут будем на каждом шаге записывать максимальное усилие, которое доступно до этой вершины и потом будем выбирать наименьшее

```python
import heapq

class Solution:
    def minimumEffortPath(self, heights: List[List[int]]) -> int:
        rows = len(heights)
        cols = len(heights[0])

        max_len = float('inf')
        q = []
        efforts = {(i, j) : max_len for i in range(rows) for j in range(cols)}

        heapq.heapify(q)
        
        offsets = [(-1, 0), (0, 1), (1, 0), (0, -1)]

        heapq.heappush(q, (0, 0, 0))

        efforts[(0,0)] = 0

        while q:
            weight, parent_row, parent_column = heapq.heappop(q)

            for off_row, off_column in offsets:

                neibor_row = parent_row + off_row
                neibor_column = parent_column + off_column

                if neibor_row >= 0 and neibor_row < rows and neibor_column >= 0 and neibor_column < cols:
                    diff = abs(heights[parent_row][parent_column] - heights[neibor_row][neibor_column])
                    
                    effort = max(weight, diff)

                    if effort < efforts[(neibor_row, neibor_column)] or efforts[(neibor_row, neibor_column)] == max_len:
                        efforts[(neibor_row, neibor_column)] = effort
                        heapq.heappush(q, (effort, neibor_row, neibor_column))

        
        return efforts[(rows - 1, cols -1)]
```
