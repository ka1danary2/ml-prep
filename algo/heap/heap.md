У кучи есть несколько важных свойств:
- значение в каждой из его вершин меньше или равно значениями в его потомках
- для каждой вершины по индексу k списка data, его два следующих потомка располагаются по индексам `2k+1`, `2k+2`

Всекие методы из бибоиотеки для работы с кучей + реализация кучи с приоритетом
```python
import heapq


d = []

# добавление элементов в итерируемый объект
heapq.heappush(d, 4)
heapq.heappush(d, 1)
heapq.heappush(d, 7)
heapq.heappush(d, 6)

print(d)

# если уже есть массив и его надо переупорядочить под кучу

data = [10, 5, 7, 16, 13, 2, 20]
heapq.heapify(data)

# наименьшее значение
min_value = data[0]
print(data)
print('min', min_value)

# извлечение из кучи минимального значения
min_value = heapq.heappop(data)

# извлечение и добвление нового элемента
min_value = heapq.heapreplace(data, 11)

# если надо заменить старое значение и добавить новое, при этом не перестраивая структуру
min_value = heapq.heappushpop(data, 1)

print(min_value)

# если надо получить несколлько минимальных значений без изменения структуры
min_values = heapq.nsmallest(3, data)

print(min_values)

# наоборот, если надо извлечь три наибольших
max_values = heapq.nlargest(3, data)
print(max_values)

# правктическое применение кучи для реализации очереди с приоритетом
class PriorityQueue:
    def __init__(self):
        self._queue = []
    
    def put(self, item, priority):
        heapq.heappush(self._queue, (priority, item))

    def get(self):
        return heapq.heappop(self._queue)
    

pq = PriorityQueue()
pq.put('Task A', 3)
pq.put('Task B', 1)
pq.put('Task C', 2)

print(pq.get())
```
