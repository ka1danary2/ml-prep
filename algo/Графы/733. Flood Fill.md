class Solution:
    def floodFill(self, image: List[List[int]], sr: int, sc: int, color: int) -> List[List[int]]:
        # нужно использовать dfs
        start_vertex_value = image[sr][sc]

        if start_vertex_value == color:
            return image

        image[sr][sc] = color

        rows = len(image) - 1
        cols = len(image[0]) - 1


        # соседи могут быть только по вертикали и только по горизонтали
        if  sr > 0 and image[sr-1][sc] == start_vertex_value :
            self.floodFill(image, sr-1, sc, color)
        if sr < rows and image[sr+1][sc] == start_vertex_value:
            self.floodFill(image, sr+1, sc, color)
        if sc > 0 and image[sr][sc-1] == start_vertex_value:
            self.floodFill(image, sr, sc-1, color)
        if sc < cols and image[sr][sc+1] == start_vertex_value:
            self.floodFill(image, sr, sc+1, color)

        return image

obj = Solution()

input_arr = [[0,0,0],[0,0,0]]

sr = 0
sc = 0
color = 0

print(obj.floodFill(input_arr, sr, sc, color))
