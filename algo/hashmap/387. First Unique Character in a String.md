```python
class Solution:
    def firstUniqChar(self, s: str) -> int:
        map = {}
        for index, el in enumerate(s):
            if el in map:
                map[el] = -1
                continue
            map[el] = index
        for el in map:
            if map[el] != -1:
                return map[el]
        return -1
```

```python
class Solution:
    def firstUniqChar(self, s: str) -> int:
        hashmap_s = {}
        for value in s:
            if value in hashmap_s:
                hashmap_s[value] = 0
                continue
            hashmap_s[value] = 1

        for key in hashmap_s:
            if key in s and hashmap_s[key] == 1:
                return s.index(key) 
        return -1
```

```python
class Solution:
    def firstUniqChar(self, s: str) -> int:
        hashmap = {}
        for index in range(len(s)):
            if not s[index] in hashmap:
                hashmap[s[index]] = index
                continue
            hashmap[s[index]] = -1
        
        for key in hashmap:
            if hashmap[key] != -1:
                return hashmap[key]
        return -1
```
