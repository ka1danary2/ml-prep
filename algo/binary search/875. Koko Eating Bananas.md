```python
import math;

def canEat(nums, k) -> int:
        total_hours_sum = 0

        for i in range(len(nums)):
            num = math.ceil(nums[i] / k)
            total_hours_sum += num
        
        return total_hours_sum
        
class Solution:
    
    def minEatingSpeed(self, piles: List[int], h: int) -> int:
        left = 1
        right = max(piles)

        while left < right:
            mid = left + (right - left) // 2

            total = canEat(piles, mid)

            if total > h:
                 left = mid + 1

            else:
                 right = mid
                    
        return right
```
