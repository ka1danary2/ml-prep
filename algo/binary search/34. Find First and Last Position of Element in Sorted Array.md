Решение
Тут надо сначала найти начальный элемент и потом конечный, либо наоборот
Использовать надо два бин поиска под каждый элемент, то есть понадобится два поиск
При нахождении индекса искомого элемента, мы не останваливаемся, а продолжаем искать слева или справа, в зависимости от того, начало или конец мы ищем

```python
class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:

        l = 0
        r = len(nums) - 1

        first = -1
        second = -1

        if len(nums) == 0:
            return [first, second]

        while (l + 1 < r):
            mid = (l + r) // 2

            if target < nums[mid]:
                r = mid

            elif target > nums[mid]:
                l = mid

            else:
                r = mid

        if nums[l] == target:
            first = l

        elif nums[r] == target:
            first = r

        l = 0
        r = len(nums) - 1

        while (l + 1 < r):
            mid = (r + l) // 2

            if nums[mid] < target:
                l = mid

            if nums[mid] > target:
                r = mid

            else:
                l = mid

        if nums[r] == target:
            second = r

        elif nums[l] == target:
            second = l

        
        return [first, second]

```
