Только тут надо посчитать количетсво клеток внутри островов

class Solution:
    def dfs(self, grid, visited, start, n, m):
        visited[start] = 1
        counter = 1

        operations = [(-1, 0), (1, 0), (0, -1), (0, 1)]
        i, j = start

        for (a, b) in operations:
            if 0 <= a + i < n and 0 <= b + j < m and visited[(a + i, b + j)] != 1 and grid[a + i][b + j] == 1:
                counter += self.dfs(grid, visited, (a + i, b + j), n, m)
        
        return counter
    
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:

        rows = len(grid)
        cols = len(grid[0])

        visited = {(i, j): 0 for i in range(rows) for j in range(cols)}
        maxArea = 0

        for i in range(rows):
            for j in range(cols):
                if grid[i][j] == 1 and visited[(i, j)] != 1:
                    c = self.dfs(grid, visited, (i, j), rows, cols)
                    maxArea = max(maxArea, c)

        return maxArea

obj = Solution()
