class Solution:
    def firstStableIndex(self, nums: list[int], k: int) -> int:
        max_left = float('-inf')
        min_temp = float('inf')

        min_arr = [0] * len(nums)

        for index in reversed(range(len(nums))):
            min_temp = min(min_temp, nums[index])
            min_arr[index] = min_temp
        
        print(min_arr)
            
        for index in range(len(nums)):
            max_left = max(max_left, nums[index])

            stable = max_left - min_arr[index]
            
            if stable <= k:
                return index
