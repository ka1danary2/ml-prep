class Solution:
    def mySqrt(self, x: int) -> int:
        l = 0
        r = x
        best = -1

        # надо вернуть квадратный корень из этого числа округденный до ближайшего числа в меньшую сторону

        while l <= r:
            mid = (l + r) // 2

            pow = mid * mid

            if pow < x:
                l = mid + 1
            
            elif pow > x:
                r = mid - 1
                best = mid

            else:
                return mid

        return best - 1
