class Solution:
    def characterReplacement(self, s: str, k: int) -> int:
        max_s = 0
        l = 0
        
        frequence = {i: 0 for i in range(26)}

        count = k

        for r in range(len(s)):
            frequence[ord(s[r]) - 65] += 1

            # если длина окна - самая частая буква в ней больше чем k, то надо сужать окно
            while r - l + 1 - max(frequence) > k:
                frequence[ord(s[l]) - 65] -= 1
                l += 1

            max_s = max(max_s, (r - l + 1))
        

        return max_s

        
inst = Solution()
print(inst.characterReplacement("ABAB", 2))
