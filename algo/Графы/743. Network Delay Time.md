```python
import heapq

class Solution:
    def networkDelayTime(self, times: List[List[int]], n: int, k: int) -> int:
        # мне нужна куча 
        # мне нужен словарь с расстояниями dict 
        # надо граф из списка вершин переделать в список смежностей

        # инициализация
            # заполнить dict inf
            # добавить начальную вершину в кучу с приоритетом 0
            # выполнять пока в куче есть элементы
            # если вес от родителя к соседу меньше чем есть в словаре, то меняем его и добавляем в кучу

            # обрабатывать пока куча не пустая

            # если в куче остались inf вершины, значит они изолированные
        graph = {}
        max_len = float('inf')
        heap = []

        heapq.heapify(heap)

        for i in range(len(times)):
            a, b, w = times[i]

            if a not in graph:
                graph[a] = []
            
            graph[a].append((b, w))

            # не надо так как граф ориентирвоанный
            if b not in graph:
                graph[b] = []
            
            # graph[b].append((a, w))

        dist = {i: max_len for i in graph}

        heapq.heappush(heap, (0, k))
        dist[k] = 0

        while heap:
            priority, heapVertex = heapq.heappop(heap)
            
            for neiborhod in graph[heapVertex]:
                vertexNew, priorityNew = neiborhod

                newWeight = priority + priorityNew
 
                if newWeight < dist[vertexNew]:
                    dist[vertexNew] = newWeight
                    heapq.heappush(heap,  (newWeight, vertexNew))

        result = max(dist.values())
        if result == max_len:
            return -1
        return result
```
