В этой задаче нужно оптимально найти центр однонаправленного связного списка
Для этого лучше всего подойдет способ [[Метод "черепаха и заяц" - медленный и быстрый указатели]]

Мой решение на питоне

```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next

class Solution:
    def deleteMiddle(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if not head:
            return head

        if not head.next:
            return head.next

        slow = head
        fast = head
        prev = None

        while fast and fast.next:
            prev = slow
            slow = slow.next
            fast = fast.next.next
        
        prev.next = prev.next.next

        return head
```
