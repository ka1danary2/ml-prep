class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        height = {}
        prefix_sum = 0
        counter = 0

        height[0] = 1

        for item in nums:
            prefix_sum += item
            print('prefix_sum ', prefix_sum)

            diff =  prefix_sum - k
            print('diff ', diff)

            if diff in height:
                print('diff in height ')
                counter += height[diff]

            if prefix_sum in height:
                print('nums[i]  in height ')
                height[prefix_sum] += 1
            else:
                print('nums[i]  not in height ')
                height[prefix_sum] = 1

            print('height ', height)
        
            print('\n')
        return counter
